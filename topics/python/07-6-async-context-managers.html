6. Async Context Managers

Async context managers work with async with and implement __aenter__ and __aexit__. They are essential for managing async resources like DB pools, HTTP sessions, and WebSocket connections.

6.1 Class-Based Async Context Manager

class AsyncDBPool:
def __init__(self, url, size=10):
self.url = url
self.size = size
async def __aenter__(self):
self.pool = await asyncpg.create_pool(self.url, min_size=self.size)
return self.pool
async def __aexit__(self, exc_type, exc_val, exc_tb):
await self.pool.close()
return False
async with AsyncDBPool('postgres://...') as pool:
async with pool.acquire() as conn:
rows = await conn.fetch('SELECT * FROM users')

6.2 Generator-Based (@asynccontextmanager)

from contextlib import asynccontextmanager
@asynccontextmanager
async def http_session():
session = aiohttp.ClientSession()
try:
yield session
finally:
await session.close()
async with http_session() as session:
resp = await session.get('https://api.example.com')
