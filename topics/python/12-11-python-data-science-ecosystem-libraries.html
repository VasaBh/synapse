11.1 NumPy Essentials

import numpy as np
a = np.array([1, 2, 3, 4])
m = np.zeros((3, 4))             # 3x4 matrix of zeros
r = np.random.randn(1000)        # 1000 random normals
# Vectorized operations (100x faster than loops)
result = a * 2 + 1               # element-wise
dot = np.dot(a, a)               # dot product
# Broadcasting
matrix = np.ones((3, 4))
matrix + np.array([1, 2, 3, 4])  # adds to each row
# Indexing
m[m > 0.5]                       # boolean indexing
m[:, 1:3]                        # slice columns

11.3 Polars (Pandas Alternative)

import polars as pl
df = pl.read_csv('data.csv')
# Lazy evaluation (query optimization)
result = (
df.lazy()
.filter(pl.col('age') > 25)
.group_by('category')
.agg(pl.col('price').mean().alias('avg_price'))
.sort('avg_price', descending=True)
.collect()  # execute optimized query plan
)
