13. Testing

13.1 pytest Fundamentals

import pytest
def test_addition():
assert 1 + 1 == 2
def test_exception():
with pytest.raises(ValueError, match='invalid'):
int('not_a_number')
# Parametrize — run test with multiple inputs
@pytest.mark.parametrize('input,expected', [
('hello', 5), ('', 0), ('python', 6),
])
def test_string_length(input, expected):
assert len(input) == expected

13.2 Fixtures

@pytest.fixture
def db_session():
session = create_test_session()
yield session
session.rollback()
session.close()
@pytest.fixture
def sample_user(db_session):
user = User(name='Alice', email='a@b.com')
db_session.add(user)
db_session.commit()
return user
def test_get_user(db_session, sample_user):
found = db_session.query(User).get(sample_user.id)
assert found.name == 'Alice'

13.3 Mocking

from unittest.mock import patch, MagicMock, AsyncMock
@patch('app.services.email.send')
def test_registration(mock_send):
register_user('alice@example.com')
mock_send.assert_called_once_with('alice@example.com', subject='Welcome!')
# Async mocking
@patch('app.client.fetch', new_callable=AsyncMock)
async def test_fetch(mock_fetch):
mock_fetch.return_value = {'status': 'ok'}
result = await get_status()
assert result == 'ok'

13.4 Async Testing

import pytest
@pytest.mark.anyio  # or @pytest.mark.asyncio
async def test_async_endpoint():
async with AsyncClient(app=app, base_url='http://test') as client:
response = await client.get('/users')
assert response.status_code == 200
assert len(response.json()) > 0
