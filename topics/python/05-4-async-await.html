4. Async / Await

Python’s async/await syntax enables writing concurrent I/O-bound code that scales to thousands of simultaneous connections on a single thread.

4.1 Coroutines & asyncio.run

import asyncio
async def fetch_data(url: str) -> dict:
async with aiohttp.ClientSession() as session:
async with session.get(url) as response:
return await response.json()
async def main():
# Run concurrently (not sequentially!)
results = await asyncio.gather(
fetch_data('https://api.example.com/users'),
fetch_data('https://api.example.com/items'),
fetch_data('https://api.example.com/orders'),
)
users, items, orders = results
asyncio.run(main())  # entry point

4.2 Tasks, gather, wait

async def main():
# Create tasks (start immediately)
task1 = asyncio.create_task(slow_operation(1))
task2 = asyncio.create_task(slow_operation(2))
# Wait for both
r1, r2 = await asyncio.gather(task1, task2)
# With timeout
try:
result = await asyncio.wait_for(slow_op(), timeout=5.0)
except asyncio.TimeoutError:
print('Operation timed out')
# Wait for first completed
done, pending = await asyncio.wait(
tasks, return_when=asyncio.FIRST_COMPLETED
)

4.3 Async Iteration & Comprehensions

# Async for loop
async for message in websocket:
process(message)
# Async comprehension
results = [await fetch(url) for url in urls]
# Async generator (see next chapter)
async for chunk in stream_data():
process(chunk)

4.4 Semaphores & Rate Limiting

sem = asyncio.Semaphore(10)  # max 10 concurrent
async def rate_limited_fetch(url):
async with sem:  # acquire slot
return await fetch(url)
# Process 1000 URLs with max 10 concurrent
results = await asyncio.gather(
*[rate_limited_fetch(url) for url in urls]
)
