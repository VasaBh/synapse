10. Duck Typing & Protocols

Python uses duck typing: if an object has the right methods, it works regardless of its class. Python 3.8+ adds Protocols for structural subtyping that works with type checkers.

10.1 Duck Typing in Practice

def process_items(container):
"""Works with any iterable — list, tuple, set, generator..."""
for item in container:  # needs __iter__
print(item)
# All of these work:
process_items([1, 2, 3])
process_items((1, 2, 3))
process_items({1, 2, 3})
process_items(range(1, 4))
process_items(x for x in [1, 2, 3])  # generator

10.2 Protocols (Structural Subtyping)

from typing import Protocol, runtime_checkable
@runtime_checkable
class Drawable(Protocol):
def draw(self, canvas: 'Canvas') -> None: ...
def bounding_box(self) -> tuple[int, int, int, int]: ...
class Circle:  # no inheritance needed!
def draw(self, canvas): canvas.draw_circle(self)
def bounding_box(self): return (self.x-self.r, self.y-self.r,
self.x+self.r, self.y+self.r)
def render(shape: Drawable):  # type-safe duck typing
shape.draw(canvas)
isinstance(Circle(), Drawable)  # True (runtime_checkable)

10.3 Abstract Base Classes (ABCs)

from abc import ABC, abstractmethod
class Shape(ABC):
@abstractmethod
def area(self) -> float: ...
@abstractmethod
def perimeter(self) -> float: ...
def describe(self):  # concrete method
return f'Area={self.area():.2f}, Perimeter={self.perimeter():.2f}'
class Circle(Shape):
def __init__(self, r): self.r = r
def area(self): return 3.14159 * self.r ** 2
def perimeter(self): return 2 * 3.14159 * self.r
