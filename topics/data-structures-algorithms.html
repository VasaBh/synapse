Section 01

Python Built-in Data Structures

Python ships with a rich set of built-in data structures that cover most real-world needs. Choosing the right one is the first performance decision you make.

list

Dynamic array. O(1) append/index, O(n) insert/search. Ordered, mutable, allows duplicates.

dict

Hash map. O(1) avg get/set/delete. Ordered (Python 3.7+), mutable, unique keys.

set / frozenset

Hash set. O(1) add/discard/lookup. Unordered, unique elements. frozenset is immutable.

tuple

Immutable sequence. O(1) index. Hashable — use as dict keys or set members.

str

Immutable Unicode sequence. O(n) concatenation — prefer join() for repeated ops.

bytes / bytearray

Immutable / mutable byte sequences. Essential for binary I/O and network protocols.

collections module — enhanced built-ins

deque

Double-ended queue. O(1) append/pop from both ends. Better than list for queue patterns.

defaultdict

dict with automatic default factory. Eliminates KeyError boilerplate.

Counter

Multiset / frequency map. most_common(), arithmetic ops, fast tallying.

OrderedDict

Remembers insertion order (mostly superseded by plain dict in 3.7+). move_to_end() useful.

namedtuple

Immutable record with named fields. Lightweight alternative to a dataclass.

ChainMap

Logical merge of multiple mappings without copying. Useful for layered config / scopes.

Arrays & Linked Lists

Python's list is a dynamic array under the hood. For true fixed-size numeric arrays use array.array or numpy.ndarray. Linked lists are rarely needed in Python but are a classic interview structure.

Dynamic Array — how Python list grows

list capacity growth (CPython): len=0 capacity=0 append → capacity jumps to 4 len=4 capacity=4 append → capacity jumps to 8 (over-allocates ~12.5%) len=8 capacity=8 append → capacity jumps to 16 ... Amortised O(1) append because reallocs are rare.

Singly Linked List

class Node:
    __slots__ = ("val", "next")   # memory-efficient
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

class LinkedList:
    def __init__(self):
        self.head = None
        self._size = 0

    def prepend(self, val):          # O(1)
        self.head = Node(val, self.head)
        self._size += 1

    def append(self, val):           # O(n)
        node = Node(val)
        if not self.head:
            self.head = node
        else:
            cur = self.head
            while cur.next:
                cur = cur.next
            cur.next = node
        self._size += 1

    def delete(self, val):           # O(n)
        dummy = Node(0, self.head)
        prev = dummy
        while prev.next:
            if prev.next.val == val:
                prev.next = prev.next.next
                self._size -= 1
                break
            prev = prev.next
        self.head = dummy.next

    def to_list(self):
        result, cur = [], self.head
        while cur:
            result.append(cur.val)
            cur = cur.next
        return result

    def __len__(self): return self._size

# usage
ll = LinkedList()
for v in [3, 1, 4, 1, 5]:
    ll.append(v)
print(ll.to_list())   # [3, 1, 4, 1, 5]
ll.delete(1)
print(ll.to_list())   # [3, 4, 1, 5]

Doubly Linked List — O(1) delete with node reference

class DNode:
    __slots__ = ("val", "prev", "next")
    def __init__(self, val=0):
        self.val = val
        self.prev = self.next = None

class DoublyLinkedList:
    def __init__(self):
        # sentinel head & tail simplify edge cases
        self.head = DNode()
        self.tail = DNode()
        self.head.next = self.tail
        self.tail.prev = self.head

    def _insert_after(self, node: DNode, val) -> DNode:
        new = DNode(val)
        new.prev, new.next = node, node.next
        node.next.prev = new
        node.next = new
        return new

    def append(self, val) -> DNode:          # O(1)
        return self._insert_after(self.tail.prev, val)

    def remove(self, node: DNode):           # O(1) given node ref
        node.prev.next = node.next
        node.next.prev = node.prev

Stacks & Queues

Stacks (LIFO) and queues (FIFO) are the backbone of DFS, BFS, undo systems, and scheduling. Python provides list for stacks and collections.deque for queues. For concurrent use, use queue.Queue.

Stack — LIFO using list

stack = []

stack.append(2)
stack.append(3)
stack[-1]         # peek  O(1) → 3
stack.pop()       # pop   O(1) → 3
bool(stack)       # False when empty

# practical: balanced parentheses
def is_balanced(s: str) -> bool:
    pairs = {')':'(', ']':'[', '}':'{'}
    stack = []
    for ch in s:
        if ch in "([{":
            stack.append(ch)
        elif ch in ")]}":
            if not stack or stack[-1] != pairs[ch]:
                return False
            stack.pop()
    return not stack

is_balanced("([{}])")   # True
is_balanced("([)]")     # False

Monotonic Stack — next greater element

def next_greater(nums: list[int]) -> list[int]:
    """For each element, find the next greater element to the right. O(n)"""
    result = [-1] * len(nums)
    stack = []          # stores indices
    for i, n in enumerate(nums):
        while stack and nums[stack[-1]] < n:
            result[stack.pop()] = n
        stack.append(i)
    return result

next_greater([2, 1, 3, 5, 4])   # [3, 3, 5, -1, -1]

BFS with Queue

from collections import deque

def bfs(graph: dict, start: str) -> list[str]:
    visited = set()
    queue = deque([start])
    order = []
    while queue:
        node = queue.popleft()
        if node in visited:
            continue
        visited.add(node)
        order.append(node)
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                queue.append(neighbor)
    return order

graph = {"A": ["B","C"], "B": ["D"], "C": ["D","E"], "D": [], "E": []}
bfs(graph, "A")   # ['A', 'B', 'C', 'D', 'E']

Hash Tables & Sets

Python's dict and set are both built on open-addressing hash tables with random probing. Average O(1) operations make them the go-to structure for frequency counting, caching, and deduplication.

How Python dict works internally

Compact table (Python 3.6+): indices: [ 6 | - | 1 | - | 3 | - | 0 ] (sparse index array) entries: [(hash0, key0, val0), (hash1, key1, val1), ...] (dense entry array) Lookup: slot = hash(key) % size → follow probe sequence until match or empty Load factor: resizes at 2/3 full (doubles capacity)
# ── Frequency counting ─────────────────────────────

# Counter (most Pythonic)

freq.most_common(3)   # [('the', 3), ...]

# defaultdict equivalent

for w in words:
    freq2[w] += 1

# ── Two-sum with hash map ───────────────────────────
def two_sum(nums: list[int], target: int) -> tuple[int,int] | None:
    seen: dict[int, int] = {}       # value → index
    for i, n in enumerate(nums):
        complement = target - n
        if complement in seen:
            return seen[complement], i
        seen[n] = i
    return None

two_sum([2, 7, 11, 15], 9)   # (0, 1)

# Fast membership over large collections

# ── LRU Cache using OrderedDict ────────────────────
from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.cache: OrderedDict[int,int] = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key: int, value: int):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.cap:
            self.cache.popitem(last=False)

lru = LRUCache(3)
lru.put(1, 10); lru.put(2, 20); lru.put(3, 30)
lru.get(1)          # 10  (moves 1 to end)
lru.put(4, 40)      # evicts key 2
lru.get(2)          # -1
Hash Collision & Performance Worst-case dict lookup is O(n) if all keys hash to the same slot. In practice this is rare unless you use unhashable or adversarial keys. Custom __hash__ should distribute bits well. Use sys.hash_info to inspect randomisation seed.

Trees & Heaps

Trees model hierarchical data. Binary search trees enable O(log n) ordered ops. Heaps power priority queues. Python's heapq module provides a min-heap over a plain list.

Binary Tree

from __future__ import annotations
from collections import deque
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class TreeNode:
    val: int
    left:  Optional["TreeNode"] = field(default=None, repr=False)
    right: Optional["TreeNode"] = field(default=None, repr=False)

# ── Build example tree ─────────────────────────────
#        4
#       / \
#      2   6
#     / \ / \
#    1  3 5  7
root = TreeNode(4,
    TreeNode(2, TreeNode(1), TreeNode(3)),
    TreeNode(6, TreeNode(5), TreeNode(7)))

# ── DFS traversals (recursive) ─────────────────────
def inorder(node):              # Left → Root → Right (sorted for BST)
    return inorder(node.left) + [node.val] + inorder(node.right) if node else []

def preorder(node):             # Root → Left → Right
    return [node.val] + preorder(node.left) + preorder(node.right) if node else []

def postorder(node):            # Left → Right → Root
    return postorder(node.left) + postorder(node.right) + [node.val] if node else []

inorder(root)   # [1,2,3,4,5,6,7]

# ── BFS / level-order traversal ────────────────────
def level_order(root) -> list[list[int]]:
    if not root:
        return []
    result, queue = [], deque([root])
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left:  queue.append(node.left)
            if node.right: queue.append(node.right)
        result.append(level)
    return result

level_order(root)   # [[4],[2,6],[1,3,5,7]]

# ── Tree height ────────────────────────────────────
def height(node) -> int:
    if not node:
        return 0
    return 1 + max(height(node.left), height(node.right))

height(root)   # 3

Binary Search Tree — insert, search, delete

class BST:
    def __init__(self):
        self.root: Optional[TreeNode] = None

    def insert(self, val: int):
        def _insert(node, val):
            if not node:
                return TreeNode(val)
            if val < node.val:
                node.left = _insert(node.left, val)
            elif val > node.val:
                node.right = _insert(node.right, val)
            return node
        self.root = _insert(self.root, val)

    def search(self, val: int) -> bool:
        cur = self.root
        while cur:
            if val == cur.val: return True
            cur = cur.left if val < cur.val else cur.right
        return False

    def delete(self, val: int):
        def _delete(node, val):
            if not node:
                return None
            if val < node.val:
                node.left = _delete(node.left, val)
            elif val > node.val:
                node.right = _delete(node.right, val)
            else:
                if not node.left:  return node.right
                if not node.right: return node.left
                # find in-order successor (min of right subtree)
                successor = node.right
                while successor.left:
                    successor = successor.left
                node.val = successor.val
                node.right = _delete(node.right, successor.val)
            return node
        self.root = _delete(self.root, val)

DFS — recursive & iterative

def dfs_recursive(graph, node, visited=None) -> list:
    if visited is None:
        visited = set()
    visited.add(node)
    result = [node]
    for neighbor in graph[node]:
        if neighbor not in visited:
            result += dfs_recursive(graph, neighbor, visited)
    return result

def dfs_iterative(graph, start) -> list:
    visited, stack, order = set(), [start], []
    while stack:
        node = stack.pop()
        if node in visited:
            continue
        visited.add(node)
        order.append(node)
        for neighbor in reversed(graph[node]):
            if neighbor not in visited:
                stack.append(neighbor)
    return order

Dijkstra's Shortest Path

import heapq

def dijkstra(graph: dict, start: str) -> dict[str, float]:
    dist = {node: float("inf") for node in graph}
    dist[start] = 0
    pq = [(0, start)]          # (distance, node)

    while pq:
        d, node = heapq.heappop(pq)
        if d > dist[node]:
            continue           # stale entry
        for neighbor, weight in graph[node]:
            nd = d + weight
            if nd < dist[neighbor]:
                dist[neighbor] = nd
                heapq.heappush(pq, (nd, neighbor))
    return dist

# {"A":0, "B":3, "C":2, "D":8}

Topological Sort — Kahn's algorithm (BFS)

def topological_sort(graph: dict[int, list[int]], n: int) -> list[int] | None:
    in_degree = [0] * n
    for u in graph:
        for v in graph[u]:
            in_degree[v] += 1

    queue = deque(node for node in range(n) if in_degree[node] == 0)
    order = []
    while queue:
        node = queue.popleft()
        order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    return order if len(order) == n else None   # None = cycle detected

dag = {0:[1,2], 1:[3], 2:[3], 3:[4], 4:[]}
topological_sort(dag, 5)   # [0, 1, 2, 3, 4] (one valid order)

Union-Find (Disjoint Set Union)

class UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank   = [0] * n

    def find(self, x: int) -> int:
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # path compression
        return self.parent[x]

    def union(self, x: int, y: int) -> bool:
        px, py = self.find(x), self.find(y)
        if px == py:
            return False                # already same component
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

    def connected(self, x: int, y: int) -> bool:
        return self.find(x) == self.find(y)

uf = UnionFind(6)
uf.union(0, 1); uf.union(1, 2); uf.union(3, 4)
uf.connected(0, 2)   # True
uf.connected(0, 3)   # False

Sorting Algorithms

Python's list.sort() and sorted() use Timsort — a hybrid merge/insertion sort that's O(n log n) worst-case and O(n) on nearly-sorted data. Understanding classic sorts is essential for interviews and edge-case optimisation.

Python's built-in sort — always prefer this

# sort() modifies in-place; sorted() returns new list
nums = [3, 1, 4, 1, 5, 9, 2, 6]
nums.sort()                              # [1,1,2,3,4,5,6,9]
nums.sort(reverse=True)                  # [9,6,5,4,3,2,1,1]

# Custom key — sort by last character
words = ["banana","apple","cherry","date"]
sorted(words, key=lambda w: w[-1])       # sort by last char

# Sort objects
from dataclasses import dataclass

    name: str
    gpa: float

Merge Sort — O(n log n) stable, divide & conquer

def merge_sort(arr: list) -> list:
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left  = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    return merge(left, right)

def merge(left: list, right: list) -> list:
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    return result + left[i:] + right[j:]

merge_sort([5, 2, 8, 1, 9, 3])   # [1,2,3,5,8,9]

Quick Sort — O(n log n) avg, O(n²) worst

import random

def quick_sort(arr: list, lo=0, hi=None) -> list:
    if hi is None:
        arr = arr[:]   # work on copy
        hi = len(arr) - 1
    if lo < hi:
        p = partition(arr, lo, hi)
        quick_sort(arr, lo, p - 1)
        quick_sort(arr, p + 1, hi)
    return arr

def partition(arr, lo, hi):
    # randomise pivot to avoid O(n²) on sorted input
    pivot_idx = random.randint(lo, hi)
    arr[pivot_idx], arr[hi] = arr[hi], arr[pivot_idx]
    pivot = arr[hi]
    i = lo - 1
    for j in range(lo, hi):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i+1], arr[hi] = arr[hi], arr[i+1]
    return i + 1

Counting Sort — O(n+k) for small integer ranges

def counting_sort(arr: list[int]) -> list[int]:
    if not arr:
        return arr
    max_val = max(arr)
    counts = [0] * (max_val + 1)
    for n in arr:
        counts[n] += 1
    result = []
    for val, cnt in enumerate(counts):
        result.extend([val] * cnt)
    return result

Searching Algorithms

Linear search is O(n). Binary search is O(log n) on sorted data — one of the most powerful techniques in an engineer's toolkit. Python's bisect module provides production-ready binary search.

Binary Search

def binary_search(arr: list, target: int) -> int:
    lo, hi = 0, len(arr) - 1
    while lo <= hi:
        mid = lo + (hi - lo) // 2   # avoids overflow vs (lo+hi)//2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1   # not found

def find_first(arr: list, target: int) -> int:
    """First occurrence of target. Returns -1 if not found."""
    lo, hi, result = 0, len(arr) - 1, -1
    while lo <= hi:
        mid = (lo + hi) // 2
        if arr[mid] == target:
            result = mid
            hi = mid - 1        # keep searching left
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return result

def find_last(arr: list, target: int) -> int:
    """Last occurrence of target."""
    lo, hi, result = 0, len(arr) - 1, -1
    while lo <= hi:
        mid = (lo + hi) // 2
        if arr[mid] == target:
            result = mid
            lo = mid + 1        # keep searching right
        elif arr[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return result

def can_ship(weights: list[int], days: int, capacity: int) -> bool:
    day, load = 1, 0
    for w in weights:
        if load + w > capacity:
            day += 1
            load = 0
        load += w
    return day <= days

def min_capacity(weights: list[int], days: int) -> int:
    """Minimum capacity ship to deliver all packages within D days."""
    lo, hi = max(weights), sum(weights)
    while lo < hi:
        mid = (lo + hi) // 2
        if can_ship(weights, days, mid):
            hi = mid
        else:
            lo = mid + 1
    return lo

def deep_size(obj, seen=None):
    if seen is None:
        seen = set()
    oid = id(obj)
    if oid in seen:
        return 0
    seen.add(oid)
    size = sys.getsizeof(obj)
    if isinstance(obj, dict):
        size += sum(deep_size(k, seen) + deep_size(v, seen) for k, v in obj.items())
    elif hasattr(obj, '__iter__') and not isinstance(obj, (str, bytes)):
        size += sum(deep_size(i, seen) for i in obj)
    return size

Space–Time tradeoffs cheat sheet

Memoisation / Caching

Trade memory for speed. Convert exponential recursion to linear with @functools.lru_cache or a memo dict.

Precomputation

Compute prefix sums, cumulative products, or sorted indices once — amortise cost over many queries.

Hash for O(1) lookup

Building a set or dict from an array costs O(n) space but drops search from O(n) to O(1).

Two Pointers

O(1) space, O(n) time on sorted arrays for pair-sum, palindrome check, container problems.

Sliding Window

O(1) space for contiguous subarray problems: max sum, longest substring without repeat.

Divide & Conquer

O(log n) levels × O(n) work per level = O(n log n). Merge sort, binary search, fast power.

Given a 1-indexed sorted array, find two numbers that add to target. Return their indices.

# Pattern: opposite-end two pointers on sorted array
# Time O(n)  Space O(1)
def two_sum_sorted(numbers: list[int], target: int) -> list[int]:
    lo, hi = 0, len(numbers) - 1
    while lo < hi:
        s = numbers[lo] + numbers[hi]
        if s == target:
            return [lo + 1, hi + 1]   # 1-indexed
        elif s < target:
            lo += 1
        else:
            hi -= 1
    return []

LC 15 · 3Sum  Medium

Find all unique triplets in an array that sum to zero.

# Pattern: sort + fix one element + two-pointer scan for the other two
# Time O(n²)  Space O(1) (output excluded)
def three_sum(nums: list[int]) -> list[list[int]]:
    nums.sort()
    result = []
    for i, a in enumerate(nums):
        if a > 0:               # all remaining are positive — stop
            break
        if i > 0 and nums[i] == nums[i-1]:
            continue            # skip duplicate fixed element
        lo, hi = i + 1, len(nums) - 1
        while lo < hi:
            s = a + nums[lo] + nums[hi]
            if s == 0:
                result.append([a, nums[lo], nums[hi]])
                lo += 1
                while lo < hi and nums[lo] == nums[lo-1]:
                    lo += 1     # skip duplicate lo
            elif s < 0:
                lo += 1
            else:
                hi -= 1
    return result

three_sum([-1, 0, 1, 2, -1, -4])   # [[-1,-1,2],[-1,0,1]]

LC 11 · Container With Most Water  Medium

Given heights of vertical lines, find two that form a container holding the most water.

# Key insight: always move the shorter side — moving the taller one
# can only decrease width without a chance to increase height.
# Time O(n)  Space O(1)
def max_area(height: list[int]) -> int:
    lo, hi = 0, len(height) - 1
    best = 0
    while lo < hi:
        water = min(height[lo], height[hi]) * (hi - lo)
        best = max(best, water)
        if height[lo] <= height[hi]:
            lo += 1
        else:
            hi -= 1
    return best

LC 42 · Trapping Rain Water  Hard

Given an elevation map, compute how much rain water can be trapped.

# Two-pointer approach: track max height seen from left & right.
# Water at position i = min(max_left, max_right) - height[i]
# Time O(n)  Space O(1)
def trap(height: list[int]) -> int:
    lo, hi = 0, len(height) - 1
    max_l = max_r = water = 0
    while lo <= hi:
        if height[lo] <= height[hi]:
            if height[lo] >= max_l:
                max_l = height[lo]
            else:
                water += max_l - height[lo]
            lo += 1
        else:
            if height[hi] >= max_r:
                max_r = height[hi]
            else:
                water += max_r - height[hi]
            hi -= 1
    return water

# Time O(n)  Space O(1)
def is_palindrome(s: str) -> bool:
    lo, hi = 0, len(s) - 1
    while lo < hi:
        while lo < hi and not s[lo].isalnum():
            lo += 1
        while lo < hi and not s[hi].isalnum():
            hi -= 1
        if s[lo].lower() != s[hi].lower():
            return False
        lo += 1; hi -= 1
    return True

Sliding Window — LeetCode Patterns

A sliding window maintains a contiguous subarray/substring by expanding the right edge and shrinking the left — replacing O(n²) nested loops with O(n). Fixed-size windows for max/min; variable-size for "longest/shortest that satisfies condition".

Variable window — expand right, shrink left when constraint violated: [a, b, c, a, b, c, b, b] L expand R until window invalid L shrink L until valid again L record best window length

LC 3 · Longest Substring Without Repeating Characters  Medium

# Pattern: variable window + set to track chars in window
# Time O(n)  Space O(min(n, alphabet))
def length_of_longest_substring(s: str) -> int:
    seen = set()
    lo = best = 0
    for hi, ch in enumerate(s):
        while ch in seen:       # shrink until no duplicate
            seen.remove(s[lo])
            lo += 1
        seen.add(ch)
        best = max(best, hi - lo + 1)
    return best

length_of_longest_substring("pwwkew")     # 3 ("wke")

LC 76 · Minimum Window Substring  Hard

# Pattern: variable window + frequency counter
# Time O(n + m)  Space O(m)   m = len(t)
from collections import Counter

def min_window(s: str, t: str) -> str:
    if not t or not s:
        return ""
    need = Counter(t)
    missing = len(t)    # chars still needed
    lo = best_lo = best_hi = 0

    for hi, ch in enumerate(s, 1):   # hi is exclusive end
        if need[ch] > 0:
            missing -= 1
        need[ch] -= 1

        if missing == 0:             # valid window found
            # shrink from left while window is still valid
            while need[s[lo]] < 0:
                need[s[lo]] += 1
                lo += 1
            if not best_hi or hi - lo < best_hi - best_lo:
                best_lo, best_hi = lo, hi
            # prepare to expand again
            need[s[lo]] += 1
            missing += 1
            lo += 1

    return s[best_lo:best_hi]

min_window("ADOBECODEBANC", "ABC")   # "BANC"

LC 239 · Sliding Window Maximum  Hard

# Pattern: monotonic deque (decreasing) — front always holds max index
# Time O(n)  Space O(k)
from collections import deque

def max_sliding_window(nums: list[int], k: int) -> list[int]:
    dq: deque[int] = deque()   # stores indices, values decreasing
    result = []
    for i, n in enumerate(nums):
        # remove indices outside window
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        # remove smaller elements — they can never be the max
        while dq and nums[dq[-1]] < n:
            dq.pop()
        dq.append(i)
        if i >= k - 1:
            result.append(nums[dq[0]])
    return result

LC 424 · Longest Repeating Character Replacement  Medium

# Key insight: window is valid if (window_size - max_freq_count) <= k
# Time O(n)  Space O(1)
def character_replacement(s: str, k: int) -> int:
    count = {}
    lo = max_freq = best = 0
    for hi, ch in enumerate(s):
        count[ch] = count.get(ch, 0) + 1
        max_freq = max(max_freq, count[ch])
        # window size - max_freq = chars we need to replace
        while (hi - lo + 1) - max_freq > k:
            count[s[lo]] -= 1
            lo += 1
        best = max(best, hi - lo + 1)
    return best

# Pattern: shrink window as soon as constraint is satisfied

def min_sub_array_len(target: int, nums: list[int]) -> int:

        while total >= target:

            lo += 1

def reorder_list(head: ListNode | None) -> None:

        fast = fast.next.next

# Monotonic increasing stack stores indices.
# When we see a shorter bar, pop taller bars and compute their max rectangles.
# Time O(n)  Space O(n)
def largest_rectangle_area(heights: list[int]) -> int:
    stack: list[int] = []   # indices of monotone-increasing bars
    max_area = 0
    heights = heights + [0]  # sentinel forces flush at end

    for i, h in enumerate(heights):
        start = i
        while stack and heights[stack[-1]] >= h:
            j = stack.pop()
            width = i - (stack[-1] + 1 if stack else 0)
            max_area = max(max_area, heights[j] * width)
            start = j
        stack.append(start)          # extend bar back to where we started popping
    return max_area

LC 150 · Evaluate Reverse Polish Notation  Medium

def eval_rpn(tokens: list[str]) -> int:
    stack: list[int] = []
    ops = {
        "+": lambda a, b: a + b,
        "-": lambda a, b: a - b,
        "*": lambda a, b: a * b,
        "/": lambda a, b: int(a / b),   # truncate toward zero

            stack.append(int(t))
    return stack[0]

LC 543 · Diameter of Binary Tree  Easy

# Diameter through a node = height(left) + height(right)
# Use a nonlocal to track global max during the height DFS.
# Time O(n)  Space O(h)
def diameter_of_binary_tree(root: TreeNode | None) -> int:
    best = 0

    def height(node: TreeNode | None) -> int:
        nonlocal best
        if not node:
            return 0
        lh = height(node.left)
        rh = height(node.right)
        best = max(best, lh + rh)   # diameter through this node
        return 1 + max(lh, rh)

    height(root)
    return best

class Codec:

            node = queue.popleft()
            if node:
                result.append(str(node.val))
                queue.append(node.left)
                queue.append(node.right)
            else:
                result.append("null")

        root = TreeNode(int(next(vals)))
        queue = deque([root])
        while queue:
            node = queue.popleft()
            for attr in ("left", "right"):
                v = next(vals)
                if v != "null":
                    child = TreeNode(int(v))
                    setattr(node, attr, child)
                    queue.append(child)
        return root

LC 72 · Edit Distance  Hard

# dp[i][j] = min edits to convert word1[:i] → word2[:j]
# match:   dp[i][j] = dp[i-1][j-1]

        for j in range(1, n + 1):

                prev if word1[i-1] == word2[j-1]
                else 1 + min(prev, dp[j], dp[j-1])
            )
    return dp[n]
