7. Async Generators & Their Uses

Async generators combine generators (yield) with async/await, enabling lazy asynchronous iteration. They are ideal for streaming data, paginated APIs, and real-time event processing.

7.1 Basic Async Generator

async def stream_pages(url):
page = 1
while True:
data = await fetch(f'{url}?page={page}')
if not data['results']:
break
for item in data['results']:
yield item  # async yield
page += 1
async for user in stream_pages('/api/users'):
process(user)

7.2 Async Generator for Real-Time Data

async def watch_events(redis_channel):
pubsub = redis.pubsub()
await pubsub.subscribe(redis_channel)
try:
async for message in pubsub.listen():
if message['type'] == 'message':
yield json.loads(message['data'])
finally:
await pubsub.unsubscribe(redis_channel)
# Server-Sent Events endpoint
async def sse_handler():
async for event in watch_events('notifications'):
yield f'data: {json.dumps(event)}\n\n'

7.3 Async Generator Cleanup

async def read_chunks(path, size=8192):
async with aiofiles.open(path, 'rb') as f:
while chunk := await f.read(size):
yield chunk
# file auto-closed when generator exits
# Force cleanup with aclose()
gen = read_chunks('large_file.bin')
async for chunk in gen:
if found_what_we_need(chunk):
await gen.aclose()  # triggers cleanup
break
