5. Context Managers

Context managers ensure resources are properly acquired and released using the with statement. They implement __enter__ and __exit__ (or use @contextmanager).

5.1 Class-Based Context Manager

class DatabaseConnection:
def __init__(self, url):
self.url = url
self.conn = None
def __enter__(self):
self.conn = connect(self.url)
return self.conn  # bound to 'as' variable
def __exit__(self, exc_type, exc_val, exc_tb):
self.conn.close()
return False  # False = re-raise exceptions
with DatabaseConnection('postgres://...') as conn:
conn.execute('SELECT ...')

5.2 Generator-Based (@contextmanager)

from contextlib import contextmanager
@contextmanager
def timer(label):
import time
start = time.perf_counter()
try:
yield  # control passes to with-block
finally:
elapsed = time.perf_counter() - start
print(f'{label}: {elapsed:.4f}s')
with timer('database query'):
results = db.execute('SELECT ...')

5.3 Combining Multiple Contexts

# Python 3.10+ parenthesized context managers
with (
open('input.txt') as src,
open('output.txt', 'w') as dst,
timer('copy'),
):
dst.write(src.read())
# contextlib.ExitStack for dynamic number of contexts
from contextlib import ExitStack
with ExitStack() as stack:
files = [stack.enter_context(open(f)) for f in filenames]
